\noindent\textbf{Problema 1.} Encuentre $n\in\N$ y $q(x)\in\Q[x]$ con $q(1)\neq0$ tales que
\begin{equation*}
    (x-1)^{n}q(x)=(x^{2}-1)(x^{3}-1)(x^{6}-5x+1)
\end{equation*}

\vspace{5mm}
\noindent\textbf{Problema 2.} Encontrar un polinomio en $\Q[x]$ no nulo tal que 
$\sqrt{1+\sqrt{2}}$ es raíz.

\vspace{5mm}
\noindent\textbf{Problema 3.} Suponga que las raices del polinomio $x^{2}+ax+b+1$ son números 
naturales. Muestre que $a^{2}+b^{2}$ no es un número primo.

\vspace{5mm}
\noindent\textbf{Problema 4.} Muestre que las funciones $sen(x), cos(x)$ y $tan(x)$ no son 
polinomios.

\vspace{1cm}
\noindent\textbf{\underline{Ejercicios Propuestos:}}

\vspace{5mm}
\noindent\textbf{Extra 1.} Sea $p(x)\in\Z[x]$ tal que toma valores $\pm1$ en $3$ enteros 
distintos. Demuestre que el polinomio no tiene raices enteras. 

\vspace{2mm}
\textbf{Hint.} Dado $p(x)$ un polinomio con coeficientes enteros, muestre que $\alpha-\beta$ 
divide a $p(\alpha)-p(\beta)$.
(\textit{Después, piense en principio del palomar})

\vspace{5mm}
\noindent\textbf{Extra 2.} Sea $p(x)$ un polinomio no constante. Mostrar que $q(x)=p(cos(x))$ no
es un polinomio.
