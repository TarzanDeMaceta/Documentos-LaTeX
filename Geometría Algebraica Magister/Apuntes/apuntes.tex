\section{Variedades Afines}
\noindent Comenzamos con nociones introductorias de geometría algebraica, aquellas que son más
cercanas a la intuición geométrica. Fijamos un cuerpo $k$ algebraicamente cerrado, usualmente
pensamos que $k=\C$.

\begin{dfn}
    El $n-$espacio afín sobre $k$, denotado por $\A_{k}^{n}$, es el conjunto de las $n-$tuplas de
    elementos de $k$. Un elemento $p=(a_{1},\dots,a_{n})\in\A_{k}^{n}$ se dice 
    punto y $a_{i}$ una coordenada de $p$.
\end{dfn}

\noindent Los espacios $\A_{k}^{n}$ y $k^{n}$, como conjuntos, son idénticos, sin embargo el 
último se piensa como un espacio vectorial sobre $k$. Por ahora, el $n-$espacio afín es un 
conjunto, que un poco más adelante dotaremos de una estructura de espacio topológico.

\begin{dfn}
    Sea $T\subseteq k[x_{1},\dots,x_{n}]$, se define el lugar de ceros de $T$ como
    \begin{equation*}
        V(T):=\{p\in\A_{k}^{n}:f(p)=0,\forall f\in T\}
    \end{equation*}
\end{dfn}

\noindent Cuando $T=\{f_{1},\dots,f_{n}\}$ simplemente se escribe $V(f_{1},\dots,f_{n})$.

\vspace{1mm}
\obs Claramente si $I=(T)$, entonces $V(I)=V(T)$. Más aún, como $k[x_{1},\dots,x_{n}]$ es un 
anillo noetheriano, todo ideal es finitamente generado. Así, $V(T)$ se puede expresar como el 
lugar de ceros de un número finito de polinomios.

\begin{dfn}
    Un conjunto $V\subseteq\A_{k}^{n}$ se dice variedad algebraica afín si $V=V(T)$ para algún 
    $T\subseteq k[x_{1},\dots,x_{n}]$.
\end{dfn}

\noindent En algunos textos lo anterior se define como conjunto algebraico, y se reserva el nombre
de variedad algebraica a un conjunto algebraico irreducible. En nuestro caso, los conjuntos 
algebraicos son variedades algebraicas.

\noindent Sea $X$ un espacio topológico, un subconjunto $Y\subseteq X$ no vacío se dice 
\emph{irreducible} si cualquier par de subconjuntos $F_{1}$ y $F_{2}$ cerrados en $Y$ tales que
\begin{equation*}
    Y=F_{1}\cup F_{2}, \hhtext{entonces se cumple que} F_{1}=Y \text{ ó } F_{2}=Y
\end{equation*}

\noindent Para definir una topología en $\A_{k}^{n}$ primero debemos exhibir los abiertos, el 
siguiente lema nos ayudará

\begin{lema}
    La unión finita de variedades algebraicas es una variedad algebraica. La intersección 
    arbitraria de variedades algebraicas es una variedad algebraica. El espacio afín y el vacío 
    son variedades algebraicas.
\end{lema}

\begin{proof}
    En primer lugar, notar que
    \begin{equation*}
        \A_{k}^{n}=V(0) \hhtext{y} \emptyset=V(1)
    \end{equation*}
    Por otro lado, si $V_{i}=V(T_{i})$ se tiene que $V_{1}\cup V_{2}=V(T_{1}T_{2})$, donde 
    $T_{1}T_{2}$ denota el conjunto de todos los productos entre elementos de $T_{1}$ y $T_{2}$.

    Sean $V_{\alpha}=V(T_{\alpha})$ una familia de variedades afines, entonces
    \begin{equation*}
        \bigcap_{\alpha} V_{\alpha}=V\left(\bigcup_{\alpha} T_{\alpha}\right)
    \end{equation*}
    Por lo tanto $\bigcap V_{\alpha}$ es una variedad algebraica.
\end{proof}

\noindent Así, el espacio afín $\A_{k}^{n}$ viene con una topología natural, llamada topología
Zariski, cuyos abiertos son el complemento de una variedad algebraica en $\A_{k}^{n}$.

\noindent En este contexto, una variedad afín se dice irreducible si lo es como subespacio de 
$\A_{k}^{n}$. Está noción se puede caracterizar en términos de ideales, para ello veamos la
siguiente definición

\newpage
\begin{dfn}
    Dado $V\subseteq\A_{k}^{n}$, definimos el ideal de $V$ en $k[x_{1},\dots,x_{n}]$, como
    \begin{equation*}
        I(V):=\{f\in k[x_{1},\dots,x_{n}]:f|_{V}\equiv0\}
    \end{equation*}
\end{dfn}

\begin{prop} \hspace{1mm}
    \begin{enumerate}
        \setlength{\itemsep}{0pt}
        \item Si $T_{1}\subseteq T_{2}$ son subconjuntos de $k[x_{1},\dots,x_{n}]$, entonces
        $V(T_{2})\subseteq V(T_{1})$.

        \item Si $X_{1}\subseteq X_{2}$ son subconjuntos de $\A_{k}^{n}$, entonces 
        $I(X_{2})\subseteq I(X_{1})$.

        \item Sea $V$ una variedad afín, entonces $V(I(V))=V$.

        \item Sean $V$ y $W$ dos variedades algebraicas en $\A_{k}^{n}$, entonces $V=W$ si y solo 
        si $I(V)=I(W)$.
    \end{enumerate}
\end{prop}

\noindent Es facil ver que $I(V)$ es un ideal, más aún, si $f^{m}\in I(V)$ entonces $f\in I(V)$, 
en otras palabras, el ideal de un conjunto es radical. No siempre tenemos que $I(V(I))=I$, el
siguiente ejemplo nos ilustra

\begin{ejm}
    Consideremos $V=V(x^{2})=\{0\}\subseteq\A_{k}^{1}$. Notemos que $I(V)=(x)$.
\end{ejm}

\noindent Surge entonces una pregunta natural: ¿Cuál es exactamente la correspondencia entre 
conjuntos de polinomios y subespacios en el espacio afín? Como veremos mas adelante, el 
Nullstellensatz nos dará una respuesta precisa, por ahora, veamos el siguiente resultado

\begin{prop}
    Una variedad afín $V\subseteq\A_{k}^{n}$ es irreducible si y solo si $I(V)$ es un ideal primo.
\end{prop}

\begin{proof}
    Supongamos que $I(V)$ no es primo, entonces existen $f_{1},f_{2}$ polinomios tales que 
    $f_{1}f_{2}\in I(V)$, pero $f_{i}\not\in I(V)$. Notar que
    \begin{equation*}
        V=(V\cap V(f_{1}))\cup(V\cap V(f_{2}))
    \end{equation*}
    pero $(V\cap V(f_{i}))\neq V$, ya que existe $p_{i}\in V$ tal que $f_{i}(p_{i})\neq0$. Luego, 
    $V$ es reducible.

    \vspace{1mm}
    Supongamos que $V$ es reducible, entonces
    \begin{equation*}
        V=V_{1}\cup V_{2}
    \end{equation*}
    con $V_{i}\neq V$ variedades afines. Entonces existe un polinomio $f_{i}$ tal que $f_{i}(p)=0$ 
    para todo $p\in V_{i}$, pero no para todo punto en $V$. Notemos que $f_{1}f_{2}\in I(V)$, pero 
    $f_{i}\not\in I(V)$. Concluimos que $I(V)$ no es primo.
\end{proof}

\begin{ejm}
    Dado que $\A_{k}^{n}=V(0)$ y $(0)$ es un ideal primo, el espacio afín es irreducible. Sin 
    embargo, la variedad afín $V=V(xy)$ es reducible, ya que $V=V(x)\cup V(y)$.
\end{ejm}

\noindent Una pregunta interesante es: ¿Por que nos interesan la variedades afines irreducibles? 
Los siguientes dos resultados nos dicen la respuesta

\begin{lema}
    Dada $\{V_{\alpha}\}_{\alpha\in A}$ una colección de variedades afines de $\A_{k}^{n}$, 
    entonces tiene un elemento minimal.
\end{lema}

\begin{proof}
    Consideremos $\mathcal{C}:=\{I(V_{\alpha})\}_{\alpha\in A}$, una colección de ideales en 
    $k[x_{1},\dots,x_{n}]$. Como este último es un anillo Noetheriano, existe un elemento maximal, 
    digamos $I(V_{\beta})$. Afirmamos que $V_{\beta}$ es minimal, de lo contrario, existe 
    $V_{\alpha}\subset V_{\beta}$, lo que implica que $I(V_{\beta})\subset I(V_{\alpha})$.
\end{proof}

\begin{teo}
    Sea $V\subseteq\A_{k}^{n}$ una variedad algebraica. Entonces existen únicas variedades afines
    irreducibles $V_{1},\dots, V_{m}$ tales que
    \begin{equation*}
        V=\bigcup V_{i} \hhtext{y} V_{i}\not\subset V_{j} \htext{para todo}i\neq j
    \end{equation*}
\end{teo}

\newpage
\begin{proof}
    Definimos
    \begin{equation*}
        \mathcal{C}:=\{
            V\subseteq\A_{k}^{n}\text{ variedad afín}:V\text{ no es unión de irreducibles}
        \}
    \end{equation*}
    Si $\mathcal{C}$ es vacío no hay nada que probar, en caso contrario, existe $V\in\mathcal{C}$
    minimal y por definición de $\mathcal{C}$ es reducible. Esto contradice que sea minimal.

    Sea $V=\bigcup V_{i}$ con $V_{i}$ irreducibles y supongamos que $V_{i}\not\subset V_{j}$ para
    todo $i\neq j$. Digamos que
    \begin{equation*}
        \bigcup V_{i}=\bigcup W_{j} 
        \htext{con}W_{i}\not\subset W_{j} \hhtext{y} V_{i},W_{j}\neq\emptyset
    \end{equation*}
    Notemos que $V_{1}=V_{1}\cap V=\bigcup(V_{1}\cap W_{j})$, como $V_{1}$ es irreducible, existe
    único $j$ tal que $V_{1}=V_{1}\cap W_{j}$, es decir, $V_{1}\subseteq W_{j}$. Del mismo modo,
    existe único $i$ tal que $W_{j}\subseteq V_{i}$, lo que implica que $V_{1}\subseteq V_{i}$,
    entonces $i=1$ y así $V_{1}=W_{j}$.
\end{proof}

\noindent Por lo tanto, toda variedad algebraica se puede descomponer como la unión finita de
variedades algebraicas irreducibles, enfocando el estudio de las variedades afines a aquellas que
son irreducibles.

\vspace{5mm}
\noindent Ahora que hemos estudiado las variedades algebraicas irreducibles, nos gustaría volver a
la definición de topología Zariski y estudiar las propiedades topológicas del $n-$espacio afín.

\vspace{1mm}
\noindent Notemos que la topología Zariski en $\A_{k}^{1}$ es equivalente a la topología cofinita,
en efecto, todo ideal en $k[x]$ es principal y por lo tanto toda variedad algebraica corresponde
con el conjunto de ceros de un polinomio $f\in k[x]$, cuyas raíces son finitas. En particular, el
espacio $\A_{k}^{1}$ no es Hausdorff.

\vspace{1mm}
\noindent El resultado anterior es más general,

\begin{prop}
    El espacio afín $\A_{k}^{n}$ no es Hausdorff.
\end{prop}

\begin{proof}
    Sean $x,y\in\A_{k}^{n}$, supongamos, por contradicción, que existen $U$ y $V$ vecindades 
    disjuntas de $x$ e $y$, respectivamente. Luego, $U^{c}$ y $V^{c}$ son variedades algebraicas
    distintas de $\A_{k}^{n}$ tales que $U^{c}\cup V^{c}=\A_{k}^{n}$. Sin embargo, el espacio afín
    es irreducible, esta contradicción prueba la afirmación.
\end{proof}

\obs Lo anterior vale para cualquier variedad afín irreducible.

\vspace{1mm}
\noindent Observemos que toda variedad algebraica irreducible $V$ es conexa, en caso contrario,
existen $U$ y $W$ abiertos disjuntos tales que $U\cup W=V$. Tomando complento tendríamos que
$V$ se escribe como la unión de dos variedades afines propias.

\begin{prop}
    Una variedad afín en $\A_{\C}^{n}$ es cerrada según la topología euclideana.
\end{prop}

\begin{proof}
    Un polinomio $f:\C^{n}\to\C$ es continua, por lo tanto, $V(f)=f^{-1}(0)\subseteq\C$ es 
    cerrado. Como la intersección arbitraria de conjuntos cerrados es cerrada, vemos que $V(T)$
    es un conjunto cerrado en $\C^{n}$.
\end{proof}

\noindent Lo anterior prueba que la topología Zariski esta contenida en la topología euclideana,
más aún, esta contención es estricta, pues la primera no es Hausdorff.

\begin{lema}
    Sea $f\in\C[x_{1},\dots,x_{n}]$ no nulo, entonces $f^{-1}(0)$ tiene interior vacío.
\end{lema}

\begin{proof}
    Procederemos por inducción en $n\in\N$. Si $n=1$, sabemos que $f$ tiene finitas raíces, en 
    caso de tener interior no vacío tendría infinitas raíces, lo que prueba el caso base. 
    Supongamos, por contradicción, que existe $U\subseteq\C^{n}$ un abierto tal que 
    $U\subseteq f^{-1}(0)$. Escribimos
    \begin{equation*}
        f=\sum_{k}f_{k}(x_{1},\dots,x_{n-1})x_{n}^{k}
        \htext{con} f_{k}\in\C[x_{1},\dots,x_{n-1}]
    \end{equation*}
    Sea $a=(a_{1},\dots,a_{n})\in U$ y consideremos la función continua
    \begin{align*}
        i:\C &\to \C^{n} \\
        z & \mapsto (a_{1},\dots,a_{n-1},z)
    \end{align*}
    Notemos que $i^{-1}(U)$ es un abierto no vacío de $\C$ y $g(z)=f(a_{1},\dots,a_{n-1},z)$ un 
    polinomio que se anula en un conjunto abierto de $\C$, por el caso base, se tiene que 
    $g\equiv0$. Luego, $f_{k}(a_{1},\dots,a_{n-1})=0$ para todo $k$.

    \vspace{1mm}
    El argumento anterior vale para todo $a\in U$ y como el mapa $a\mapsto(a_{1},\dots,a_{n-1})$ 
    es abierto, el polinomio $f_{k}$ se anula en un conjunto abierto y por hipotesis inductiva 
    $f_{k}\equiv0$ para todo $k$. Concluimos que $f^{-1}(0)$ tiene interior vacío.
\end{proof}

\noindent El lema implica inmediatamente lo siguiente

\begin{cor}
    Sea $f\in\C[x_{1},\dots,x_{n}]$ no constante, entonces $f^{-1}(\C\setminus\{0\})$ es denso en 
    $\C^{n}$.
\end{cor}

\noindent Dada $V\subseteq\A_{k}^{n}$ una variedad algebraica afín propia, entonces $V=V(I)$ con 
$I\subseteq\C[x_{1},\dots,x_{n}]$ un ideal no nulo. Por el teorema de la base de Hilbert existen 
$f_{1},\dots,f_{m}$ tales que $I=(f_{1},\dots,f_{m})$, así 
\begin{equation*}
    V=V(I)=V(f_{1},\dots,f_{m})=\bigcap V(f_{i})\subseteq V(f_{1})
\end{equation*}
Tomando complemento y luego clausura, se obtiene que $V^{c}$ es denso en $\C^{n}$, es decir, todo 
abierto en la topología Zariski es denso según la topología euclideana.

\subsection{Morfismos de Variedades algebraicas}
\noindent Ahora que hemos estudiado las variedades afines, nos gustaría -- pensando en teoría de
categorías -- definir una noción de morfismo entre variedades algebraicas. Comenzamos dando la 
siguiente definición auxiliar

\begin{dfn}
    Un mapa $f:\A_{k}^{n}\to\A_{k}^{m}$ se dice polinomial si viene dado por polinomios en cada
    coordenada.
\end{dfn}

\begin{ejm}
    Considerar $f:\A_{k}^{3}\to\A_{k}^{2}$ dada por
    \begin{equation*}
        f(x,y,z)=(xy^{2}+1,xz)
    \end{equation*}
\end{ejm}

\begin{dfn}
    Un cambio de coordenadas afín en $\A_{k}^{n}$ es una aplicación polinomial 
    \begin{equation*}
        \varphi=(f_{1},\dots,f_{n}):\A_{k}^{n}\to\A_{k}^{n}
    \end{equation*}
    tal que $gr(f_{i})=1$ y $\varphi$ es invertible.
\end{dfn}

\begin{prop}
    Un mapa polinomial $f:\A_{k}^{n}\to\A_{k}^{m}$ es continuo bajo la topología Zariski.
\end{prop}

\begin{proof}
    Sea $f=(f_{1},\dots,f_{r}):\A_{k}^{n}\to\A_{k}^{m}$, donde 
    $f_{1},\dots,f_{r}\in k[x_{1},\dots,x_{n}]$. Basta probar que la preimagen de una variedad 
    afín es una variedad afín.

    Sea $V$ una variedad algebraica, entonces
    \begin{equation*}
        V=V(T) \htext{para algún} T\subseteq k[x_{1},\dots,x_{m}]
    \end{equation*}
    Así,
    \begin{equation*}
        f^{-1}(V)=\{x\in\A_{k}^{n}:g(f(x))=0\text{ para todo }g\in T\}
    \end{equation*}
    Para todo $g\in T$, la composición $g\circ f=g(f_{1},\dots,f_{r})$ es un polinomio. Por lo 
    tanto
    \begin{equation*}
        f^{-1}(V)=V(\{g(f_{1},\dots,f_{r}):g\in T\})
    \end{equation*}
    Lo que concluye la demostración.
\end{proof}

\noindent En general, consideramos la siguiente definición

\begin{dfn}
    Sean $V\subseteq\A_{k}^{n}$ y $W\subseteq\A_{k}^{n}$ variedades afines. Una función 
    $\varphi:V\to W$ se dice morfismo de variedades afines si es la restricción de un mapa 
    polinomial.
\end{dfn}

\noindent Veamos un ejemplo

\begin{ejm}
    Consideremos $V=V(x^{2}-y)$, el mapa $\varphi:\C\to V$ dado por
    \begin{equation*}
        \varphi(z)=(z,z^{2})
    \end{equation*}
    es un morfismo de variedades afines, de hecho, es un isomorfismo.
\end{ejm}

\noindent Definimos $AffVar$ como la categoría cuyos objetos son las variedades algebraicas y sus
morfismos, los morfismos de variedades algebraicas.

\vspace{1mm}
\obs Dado que un mapa polinomial es continuo según la topología Zariski, un morfismo de variedades 
algebraicas es continuo en el sentido Zariski. En términos categóricos, tenemos un funtor 
covariante, inyectivo y no sobreyectivo
\begin{equation*}
    \textbf{AffVar}\hookrightarrow \textbf{Top}
\end{equation*}

\subsection{Nullstellensatz}

En esta sección daremos una biyección explícita entre las variedades algebraicas y los ideales
radicales en $k[x_{1},\dots,x_{n}]$. Empezaremos enunciando, sin demostrar, el siguiente lema que
no será de gran utilidad

\begin{lema}[Lema de Zariski]
    Sea $K\subseteq L$ una extensión de cuerpos tal que $L$ es finitamente generado como 
    $K-$álgebra, entonces $L$ es finitamente generado como $K-$módulo.
\end{lema}

\noindent Con el lema anterior podemos demostrar el siguiente resultado

\begin{lema}
    Sea $I\subset k[x_{1},\dots,x_{n}]$ un ideal, entonces $V(I)\neq\emptyset$
\end{lema}

\begin{proof}
    Sin perdida de generalidad podemos suponer que $I$ es maximal, en caso contrario, existe un
    ideal maximal $J$ tal que $I\subseteq J$, y por lo tanto $V(J)\subseteq V(I)$. De este modo,
    \begin{equation*}
        L=k[x_{1},\dots,x_{n}]/I
    \end{equation*}
    es un cuerpo que contiene a $k$. Por el lema anterior, $L$ es un $k-$módulo finitamente 
    generado, por ende, es una extensión algebraica de $k$, como este último es algebraicamente
    cerrado, tenemos que $L=k$.

    Así, para cada $x_{i}$ existe $a_{i}\in k$ tal que $x_{i}-a_{i}\in I$, es decir, 
    $I=(x_{1}-a_{1},\dots,x_{n}-a_{n})$. Concluimos que $V(I)\neq\emptyset$.
\end{proof}

\noindent El lema previo, junto con el truco de Rabinowitsch, nos permite demostrar el siguiente
teorema, también conocido como Nullstellensatz

\begin{teo}[\textbf{Nullstellensatz}]
    Sea $I\subset k[x_{1},\dots,x_{n}]$ un ideal. Consideremos $V=V(I)\subseteq\A_{k}^{n}$ la
    variedad afín asociada a $I$. Entonces, $I(V)=\sqrt{I}$. En otras palabras, hay una
    correspondencia
    
    \begin{equation*}
        \begin{Bmatrix}
            \textit{Variedades afines} \\
            \textit{en }\A_{k}^{n}
        \end{Bmatrix}
        \longleftrightarrow
        \begin{Bmatrix}
            \textit{Ideales radicales} \\
            \textit{en }k[x_{1},\dots,x_{n}]
        \end{Bmatrix}
    \end{equation*}
\end{teo}

\vspace{2mm}
\begin{proof}
    Supongamos que $f\in\sqrt{I}$, por definición existe $n\in\N$ tal que $f^{n}\in I$. Luego,
    $f^{n}(p)=0$, por ende $f(p)=0$, para todo $p\in V$, lo que implica que $f\in I(V)$.

    \vspace{1mm}
    Sea $f\in I(V)$ y digamos que $I=(f_{1},\dots,f_{m})$. Definimos el ideal
    \begin{equation*}
        J:=(f_{1},\dots,f_{m},x_{n+1}f-1)\subseteq k[x_{1},\dots,x_{n+1}]
    \end{equation*}
    Supongamos que existe $a\in V(J)$, en particular, se tiene que $f_{i}(a)=0$ para todo $i$, 
    entonces $a\in V$ y por lo tanto $f(a)=0$. Sin embargo, notamos lo siguiente
    \begin{equation*}
        0=a_{n+1}f(a)-1=-1
    \end{equation*}
    Concluimos que $V(J)=\emptyset$. Por lo tanto, tenemos que $J=k[x_{1},\dots,x_{n+1}]$. De este
    modo $g_{1},\dots,g_{m+1}\in k[x_{1},\dots,x_{n+1}]$ tales que
    \begin{equation*}
        g_{1}f_{1}+\cdots,g_{m}f_{m}+g_{m+1}(x_{n+1}f-1)=1
    \end{equation*}
    y tomando $x_{n+1}=1/f$ obtenemos que
    \begin{equation*}
        g_{1}(x_{1},\dots,x_{n},1/f)f_{1}+\cdots+g_{m}(x_{1},\dots,x_{n},1/f)f_{m}=1
    \end{equation*}
    Así, existe $n\in\N$ de modo que $f^{n}\in I$.
\end{proof}

\noindent Para probar la contención $I(V)\subseteq\sqrt{I}$ usamos lo que se conoce como el truco 
de Rabinowitsch.

\begin{ejm}
    Veamos un ejemplo donde el Nullstellensatz puede fallar si no pedimos que $k$ sea 
    algebraicamente cerrado. Considerar $V=V(x^{3}+x)\subseteq\A_{\R}^{1}$. Notemos que $V=\{0\}$,
    por lo tanto $I(V)=(x)$, pero $x\not\in\sqrt{(x^{3}+x)}$, ya que $x^{2}+1$ y $x$ son coprimos.
\end{ejm}

\noindent Hay resultados interesantes derivados del Nullstellensatz.

\begin{cor}
    Los ideales maximales en $k[x_{1},\dots,x_{n}]$ se corresponden con puntos en $\A_{k}^{n}$, y
    los ideales primos se corresponden con variedades algebraicas irreducibles.
\end{cor}

\begin{cor}
    Sea $f\in k[x_{1},\dots,x_{n}]$ un polinomio irreducible, entonces $I(V(f))=(f)$.
\end{cor}

\begin{proof}
    Como $f$ es irreducible, es primo, por lo tanto el ideal $(f)$ también es primo.
\end{proof}

\noindent Vimos un ejemplo donde Nullstellensatz falla si no pedimos que $k$ sea algebraicamente
cerrado. Sin embargo, la situación puede ser más catastrófica, ya que toda variedad algebraica es
el conjunto de ceros de una función en un cuerpo no algebraicamente cerrado. Usaremos el siguiente
lema

\begin{lema}
    Sea $k$ no algebraicamente cerrado, entonces existe un polinomio $q\in k[x,y]$ tal que 
    $q(x,y)=0$ si y solo si $x=y=0$.
\end{lema}

\begin{proof}
    Como $k$ no es algebraicamente cerrado, existe $f\in k[x]$ un polinomio no constante sin 
    raíces en $k$. Escribamos
    \begin{equation*}
        f=a_{0}+a_{1}x+\cdots+a_{n}x^{n} \htext{con}a_{0},a_{n}\neq0
    \end{equation*}
    Definimos $q(x,y)\in k[x,y]$ como
    \begin{equation*}
        q(x,y):=\widetilde{f}=a_{0}y^{n}+a_{1}xy^{n-1}+\cdots+a_{n-1}x^{n-1}y+a_{n}x^{n}
    \end{equation*}
    Sea $(x,y)\in k^{2}$ tal que $q(x,y)=0$. Supongamos que $y\neq0$, de lo contrario, 
    $a_{n}x^{n}=0$ y entonces $x=0$. Luego,
    \begin{equation*}
        0=a_{0}+a_{1}\left(\frac{x}{y}\right)+\cdots+a_{n}\left(\frac{x}{y}\right)^{n}
        =f\left(\frac{x}{y}\right)
    \end{equation*}
    Pero la ecuación anterior no tiene solución en $k$ por la elección de $f$. Concluimos que
    $q(x,y)=0$ si y solo si $x=y=0$.
\end{proof}

\vspace{1mm}
\noindent Dada $V=V(f_{1},\cdots,f_{m})\subseteq\A_{k}^{n}$ una variedad algebraica, tomar 
$q\in k[x,y]$ como en el lema, definimos recursivamente los siguientes polinomios
\begin{equation*}
    q_{i}:=q(f_{i},q_{i+1}) \htext{para}1\leq i<m
\end{equation*}
y $q_{m}=f_{m}$. Afirmamos que $V(q_{1})=V$, en efecto, se tiene que $q_{1}=q(f_{1},q_{2})=0$ si y 
solo si $f_{1}=q_{2}=0$, del mismo modo, $q_{2}=0$ equivale a que $f_{2}=q_{3}=0$ e inductivamente 
tenemos que $q_{1}=0$ si y solo si $f_{i}=0$ para todo $i$, lo que prueba la afirmación.

\subsection{Anillo de Coordenadas en Variedades afines}

\noindent Ahora que sabemos que hay una biyección entre variedades afines e ideales radicales, nos
interesa estudiar la estructura algebraica asociada a una variedad afín.

\begin{dfn}
    Sea $V\subseteq\A_{k}^{n}$ una variedad afín. Una función $f:V\to k$ se dice regular si existe
    un polinomio $\tilde{f}\in k[x_{1},\dots,x_{n}]$ tal que $f=\tilde{f}|_{V}$. El conjunto de 
    funciones regulares sobre $V$ induce una $k-$álgebra denotada por $k[V]$.
\end{dfn}

\noindent A la $k-$álgebra $k[V]$ se le dice anillo de coordenadas de $V$.

\begin{prop}
    Existe un isomorfismo de $k-$álgebras
    \begin{equation*}
        k[V]\cong k[x_{1},\dots,x_{n}]/I(V)
    \end{equation*}
\end{prop}

\begin{proof}
    Definimos $\varphi:k[V]\to k[x_{1},\dots,x_{n}]/I(V)$ como
    \begin{equation*}
        \varphi(f)=[\tilde{f}]
    \end{equation*}
    Claramente es un morfismo de $k-$álgebras sobreyectivo. Veamos que es inyectivo. En efecto,
    sean $f_{1},f_{2}\in k[x_{1},\dots,x_{n}]$ tales que $f=\tilde{f_{i}}|_{V}$, entonces
    \begin{equation*}
        \tilde{f_{1}}|_{V}-\tilde{f_{2}}|_{V}\equiv0,
        \htext{equivalentemente } f_{1}-f_{2}\in I(V)
    \end{equation*}
    Lo que concluye la demostración.
\end{proof}

\obs Una $k-$álgebra se dice reducida, si no posee elementos nilpotentes. La proposición anterior 
nos dice que el anillo de coordenadas de $V$ es un álgebra reducida. No es díficil verificar lo
siguiente

\begin{lema}
    Sea $I\subseteq A$ un ideal, donde $A$ es un anillo conmutativo con unidad. Entonces $I$ es 
    reducido si y solo si $A/I$ es un anillo reducido.
\end{lema}

\noindent Este lema nos ayudará para el siguiente teorema

\begin{teo}
    Sea $R$ una $k-$álgebra reducida y finitamente generada. Entonces $R\cong k[V]$ para alguna
    variedad afín $V$.
\end{teo}

\begin{proof}
    Supongamos que $R$  esta generado por $r_{1},\dots,r_{n}$. Definimos el mapa
    \begin{align*}
        f:k[x_{1},\dots,x_{n}] &\to R \\
        x_{i} &\mapsto r_{i}
    \end{align*}
    Que resulta ser un morfismo de $k-$álgebras sobreyectivo, luego 
    $R\cong k[x_{1},\dots,x_{n}]/\kr{f}$ y como $R$ es reducible tenemos que 
    $\kr{f}=\sqrt{\kr{f}}=I(V(\kr{f}))$.
\end{proof}

\noindent Lo anterior nos da una correspondencia entre $k-$álgebras reducidas y finitamente 
generadas, y variedades afines. Más aún, esta correspondencia es funtorial.

\noindent Sean $V\subseteq \A_{k}^{n}$ y $W\subseteq \A_{k}^{m}$ variedades algebraicas. Dado 
$\varphi: V\to W$ un morfismo de variedades afines, induce un morfismo de $k-$álgebras dado por
\begin{align*}
    \varphi^{*}:k[W] &\to k[V] \\
    f &\mapsto f\circ\varphi
\end{align*}
Por otro lado, dado un morfismo de $k-$álgebras 
$\sigma:k[x_{1},\dots,x_{m}]/J\to k[x_{1},\dots,x_{n}]/I$ induce un mapa polinomial dado por
\begin{equation*}
    \varphi(\sigma)=(f_{1},\dots,f_{m}):V(I)\to V(J)
\end{equation*}
donde $f_{i}\in k[x_{1},\dots,x_{n}]$ cumple que $\sigma(x_{i})=f_{i}$ para todo $1\leq i\leq m$.
Debemos verificar que $\varphi(\sigma)$ esta bien definida. Notemos que, dado $g\in J$, se tiene 
que $g(f_{1},\dots,f_{m})=\sigma(g)\in I$ porque $\sigma$ es morfismo. Por lo tanto, si 
$a\in V(I)$, la observación anterior implica que $\varphi(\sigma)(a)\in V(J)$.

\vspace{1mm}
\noindent Esta definición es independiente de la elección de $f_{i}$. En efecto, si 
$f_{i}, g_{i}$ son tales que $f_{i}=g_{i}=\sigma(x_{i})$, entonces $f_{i}-g_{i}\in I$ y por ende
$f_{i}(a)=g_{i}(a)$ para todo $a\in V(I)$. Tenemos una equivalencia de categorías
\begin{equation*}
    \textbf{AffVar}_{k}\simeq \textbf{Alg}_{k}^{red,fg}
\end{equation*}

\begin{ejm}
    Consideremos la variedad afín $V=V(y-x^{2},z-x^{3})\subseteq\A_{k}^{3}$. El morfismo
    $\varphi:\A_{k}^{1}\to V$ de variedades algebraicas dado por
    \begin{equation*}
        t\mapsto (t,t^{2},t^{3})
    \end{equation*}
    induce un morfismo de $k-$álgebras $\varphi^{*}:k[x,y,z]/(y-x^{2},z-x^{3})\to k[t]$ tal que
    \begin{gather*}
        x\mapsto t \qquad y\mapsto t^{2} \qquad z\mapsto t^{3}
    \end{gather*}
    Luego, $\varphi(\varphi^{*})=(t,t^{2},t^{3})$.
\end{ejm}

\section{Variedades Proyectivas} %% Clase 4 y 5

\noindent Hemos explorado y discutido propiedades topológicas, geométricas y algebraicas de las
variedades afines. Ahora, nos gustaría estudiar, con un enfoque similar, las variedades 
proyectivas. Al igual que antes, comenzamos fijando un cuerpo $k$ algebraicamente cerrado.

\begin{dfn}
    Definimos el $n-$espacio proyectivo sobre $k$ como
    \begin{equation*}
        \P_{k}^{n}:=\left(\A_{k}^{n+1}\setminus\{0\}\right)/\sim
    \end{equation*}
    donde $x\sim y$ si y solo si existe $\lambda\in k^{*}$ tal que $\lambda x=y$. Un elemento en
    $\P_{k}^{n}$ se dice punto y un representante de la clase se dice coordenada homogénea.
\end{dfn}

\obs Un punto en $\P_{k}^{n}$ típicamente se escribe por algún representante de la clase y la 
notación usual es $[x_{0}:\dots:x_{n}]$, donde $x_{i}\neq0$ para algún $i$.

\vspace{1mm}
\noindent Notemos que $k^{*}$ es un grupo que actúa naturalmente en $\A_{k}^{n+1}$ mediante 
multiplicación por escalar. Y la órbita de un punto corresponde a una recta en $\A_{k}^{n+1}$, es
decir, podemos identificar el espacio proyectivo con el espacio de rectas que pasan por $0$ en el
espacio afín.

\vspace{1mm}
\noindent Adicionalmente, es posible identificar el espacio afín con una carta afín de 
$\P_{k}^{n}$ mediante la asignación
\begin{equation*}
    (x_{1},\dots,x_{n})\mapsto[1:x_{1}:\dots:x_{n}]
\end{equation*}
Donde una carta afín de $\P_{k}^{n}$ se define como
\begin{equation*}
    U_{i}:=\left\{[x_{0}:\dots:x_{n}]\mid x_{i}=1\right\}
    =\left\{[x_{0}:\dots:x_{n}]\mid x_{i}\neq0\right\}
\end{equation*}
Por definición, tenemos que $\P_{k}^{n}=\bigcup_{i} U_{i}$. La correspondencia anterior no es 
única; por cada carta afín tenemos un mapa que la identifica con $\A_{k}^{n}$ y cada una de ellas
nos da una perspectiva de $\P_{k}^{n}$.

\vspace{1mm}
\noindent Para definir una variedad proyectiva, primero, debemos darle sentido a la noción de ser 
cero de un polinomio; y estos últimos, para respetar la estructura, serán homogéneos

\begin{dfn}
    Un polinomio $f\in k[x_{0},\dots,x_{n}]$ se dice homogéneo de grado $d>0$ si todos los 
    términos de $f$ son monomios de grado $d$.
\end{dfn}

\noindent Equivalentemente, un polinomio $f$ es homogéneo si y solo si para todo 
$\lambda\in k^{*}$ se tiene que
\begin{equation*}
    f(\lambda x_{0},\dots,\lambda x_{n})=\lambda^{d}f(x_{0},\dots,x_{n})
\end{equation*}

\begin{dfn}
    Decimos que $p\in\P_{k}^{n}$ es cero de un polinomio homogéneo $f\in k[x_{0},\dots,x_{n}]$ si
    existen coordenadas homogéneas $[x_{0}:\dots:x_{n}]$ tales que $f(x_{0},\dots,x_{n})=0$. 
    Escribimos $f(p)=0$.
\end{dfn}

\obs La propiedad de ser cero no depende del representante.

\begin{dfn}
    Sea $T\subseteq k[x_{0},\dots,x_{n}]$ un conjunto de polinomios homogéneos, definimos el 
    lugar de ceros de $T$ como
    \begin{equation*}
        V^{p}(T):=\{p\in\P_{k}^{n}:f(p)=0, \forall f\in T\}
    \end{equation*}
    Un conjunto $V\subseteq\P_{k}^{n}$ se dice variedad proyectiva si existe $T$ tal que 
    $V=V^{p}(T)$.
\end{dfn}

\begin{ejm}
    Consideremos la variedad proyectiva $V=V^{p}(x^{2}+y^{2}-z^{2})\subseteq\P_{k}^{2}$. Si 
    tomamos $[x:y:z]\in V\cap U_{2}$, sin perdida de generalidad, podemos asumir que $z=1$ y como 
    es cero de $x^{2}+y^{2}-z^{2}$ se cumple que
    \begin{equation*}
        x^{2}+y^{2}=1
    \end{equation*}
    y bajo la identificación, esta variedad proyectiva se corresponde con $V(x^{2}+y^{2}-1)$ en 
    $\A_{k}^{2}$. Por el mismo razonamiento, el conjunto $V\cap U_{i}$ se indentifica con 
    $V(x^{2}-y^{2}+1)$, donde $i=0,1$.
\end{ejm}

\noindent Como en el caso afín, el espacio proyectivo también admite una topología Zariski, cuyos 
abiertos corresponden al complemento de una variedad proyectiva.

\vspace{1mm}
\obs La multiplicación de polinomios homogéneos es un polinomio homogéneo cuyo grado es la suma 
estos polinomios homogéneos. Luego, la unión finita de variedades proyectivas es, denuevo, una 
variedad proyectiva.

\begin{prop}
    Los mapas
    \begin{align*}
        \psi_{i}:U_{i} &\to \A_{k}^{n} \\
        [a_{0}:\dots:a_{i-1}:1:a_{i+1}:\dots:a_{n}] &\mapsto 
        (a_{0},\dots,a_{i-1},a_{i+1},\dots,a_{n})
    \end{align*}
    son homeomorfismos respecto a la topología Zariski.
\end{prop}

\begin{proof}
    Veamos que $\psi$ es una biyección, en efecto, su inversa esta dada por
    \begin{align*}
        \phi_{i}:\A_{k}^{n} &\to U_{i} \\
        (a_{0},\dots,a_{n}) &\mapsto [a_{0}:\dots:a_{i-1}:1:a_{i}:\dots:a_{n}]
    \end{align*}
    Es inmediato que $\psi_{i}\circ\phi_{i}=id_{\A_{k}^{n}}$ y $\phi_{i}\circ\psi_{i}=id_{U_{i}}$.
    Resta ver que $\psi_{i}$ y $\phi_{i}$ son continuas.

    \vspace{1mm}
    Sea $V(f)\subseteq\A_{k}^{n}$ donde $f\in k[x_{0},\dots,\hat{x_{i}},\dots,x_{n}]$ (Un 
    polinomio en $k[x_{0},\dots,x_{n}]$ que no contiene a la variable $x_{i}$). Sea $d=deg(f)$, 
    definimos
    \begin{equation*}
        F(x_{0},\dots,x_{n})
        =x_{i}^{d}f\left(\frac{x_{0}}{x_{i}},\dots,\frac{x_{n}}{x_{i}}\right)
        \in k[x_{0},\dots,x_{n}]
    \end{equation*}
    Notar que $F$ es homogéneo de grado $d$ y además
    \begin{equation*}
        \psi_{i}^{-1}(V(f))=U_{i}\cap V(F)
    \end{equation*}
    Por lo tanto, $\psi_{i}$ es continuo. Por otro lado, sea $U_{i}\cap V(F)$ donde 
    $F\in k[x_{0},\dots,x_{n}]$ es homogéneo. Entonces
    \begin{equation*}
        \phi_{i}^{-1}(U_{i}\cap V(F))=V(f) 
        \hhtext{donde}
        f=F(x_{0},\dots,x_{i-1},1,x_{i+1},\dots,x_{n})
    \end{equation*}
    Luego, $\phi_{i}$ es continua. Concluimos que $\psi_{i}$ es un homeomorfismo.
\end{proof}

\obs Dado $p\in U_{i}$ existe un único representante $[x_{0}:\dots:x_{n}]$ tal que $x_{i}=1$, y 
por ende $\psi_{i}$ está bien definida.

\begin{cor}
    El $n-$espacio proyectivo $\P_{k}^{n}$ no es Hausdorff.
\end{cor}

\noindent Nuestro objetivo ahora es el Nullstellensatz proyectivo, comenzamos notando que 
$k[x_{0},\dots,x_{n}]$ se puede escribir como $\bigoplus S_{d}$, donde $S_{d}$ es el grupo 
abeliano de polinomios homogéneos de grado $d$.

\begin{dfn}
    Un ideal $I\subseteq k[x_{0},\dots,x_{n}]$ se dice homogéneo si
    \begin{equation*}
        I=\bigoplus (I\cap S_{d})
    \end{equation*}
\end{dfn}

\noindent Equivalentemente, $I$ es un ideal homogéneo si y solo si está generado por polinomios
homogéneos.

\begin{ejm}
    El ideal $I=(x^{2}+y)\subseteq k[x,y]$ no es homogéneo porque $y\not\in I$. Sin embargo, el 
    ideal
    \begin{equation*}
        I\subseteq (x^{2},y)\subseteq k[x,y]
    \end{equation*}
    si es homogéneo. 
\end{ejm}

\newpage
\begin{prop}
    Sea $I\subseteq k[x_{0},\dots,x_{n}]$ un ideal. Entonces $I$ es homogéneo si y solo si 
    contiene todos los componentes homogéneos de sus elementos.
\end{prop}

\begin{proof}
    Supongamos que $I$ es homogéneo, entonces $I$ esta generado por polinomios homogéneos, digamos
    $I=\gen{\{g_{\alpha}\}_{\alpha\in A}}$, donde $g_{\alpha}$ es homogéneo. Sea $f\in I$, luego
    \begin{equation*}
        f=\sum_{\alpha}h_{\alpha}g_{\alpha} \htext{con} h_{\alpha}=0 \text{ salvo finitos}
    \end{equation*}
    Escribimos $h_{\alpha}$ como suma de sus componentes homogéneas
    \begin{equation*}
        h_{\alpha}=\sum_{j}(h_{\alpha})_{j}
    \end{equation*}
    Si $deg(g_{\alpha})=d_{\alpha}$, entonces la componente homogénea de grado $r$ de $f$ esta 
    dada por
    \begin{equation*}
        f_{r}=\sum_{\alpha}(h_{\alpha})_{r-d_{\alpha}}g_{\alpha}
        \htext{donde} (h_{\alpha})_{j}=0 \text{ si } j<0
    \end{equation*}
    Concluimos que $f_{r}\in I$ para todo $r$. Por otro lado, supongamos que $I$ contiene todos
    los componentes homogéneos de sus elementos. 
    
    \vspace{1mm}
    Dado $f\in I$ vemos que $f=f_{0}+\dots+f_{d}$ donde $f_{i}$ es homogéneo de grado $i$, 
    entonces $f_{i}\in I$. Por lo tanto, $I$ esta generado por polinomios homogéneos.
\end{proof}

\begin{lema}
    Sea $I\subseteq k[x_{0},\dots,x_{n}]$ un ideal homogéneo, entonces $\sqrt{I}$ es homogéneo.
\end{lema}

\begin{proof}
    Sea $f\in\sqrt{I}$, escribimos
    \begin{equation*}
        f=f_{r}+f_{r+1}+\dots+f_{d}
        \htext{donde}f_{r}\neq0 \text{ y $f_{i}$ homogéneo de grado }i
    \end{equation*}
    Por la proposición anterior basta probar que $f_{i}\in\sqrt{I}$ para todo $i$. Como 
    $f\in\sqrt{I}$, existe $n\in\N$ tal que $f^{n}\in I$. El componente homogéneo de menor grado
    de $f^{n}$ es $f_{r}^{n}$, y dado que $I$ es homogéneo, se tiene que
    \begin{equation*}
        f_{r}^{n}\in I \hhtext{entonces} f_{r}\in\sqrt{I}
    \end{equation*}
    Por lo tanto $f-f_{r}=f_{r+1}+\dots+f_{d}\in\sqrt{I}$, inductivamente, vemos que 
    $f_{i}\in\sqrt{I}$ para todo $i$, lo que prueba lo pedido.
\end{proof}

\begin{dfn}
    Sea $I\subseteq k[x_{0},\dots,x_{n}]$ un ideal homogéneo, definimos
    \begin{equation*}
        V^{p}(I)=\{p\in\P_{k}^{n}:f(p)=0 \text{ para todo } f\in I \text{ homogéneo}\}
    \end{equation*}
\end{dfn}

\obs Como $I$ es homogéneo tenemos que $I=\gen{\{g_{\alpha}\}_{\alpha\in A}}$, donde $g_{\alpha}$ 
es homogéneo. Si consideramos $T=\{g_{\alpha}\}_{\alpha\in A}$, entonces $V^{p}(I)=V^{p}(T)$.

\vspace{1mm}
\noindent Consideremos el ideal maximal y homogéneo 
$\m=(x_{0},\dots,x_{n})\subseteq k[x_{0},\dots,x_{n}]$, como el único punto en $\A_{k}^{n+1}$ que 
se anula en $\m$ es $0$, tenemos que $V^{p}(\m)=\emptyset$.

\begin{dfn}
    Sea $V\subseteq\P_{k}^{n}$ una variedad algebraica, se define
    \begin{equation*}
        I^{h}(V)=\gen{\{f\in k[x_{0},\dots,x_{n}] \text{ homogéneo}:f|_{V}\equiv0\}}
    \end{equation*}
\end{dfn}

\obs Claramente $I^{h}(V)$ es un ideal homogéneo y radical. Además, una variedad proyectiva $V$ es 
irreducible, como subespacio de $\P_{k}^{n}$, si y solo si $I^{h}(V)$ es un ideal primo.

\newpage

\begin{dfn}
    Sea $Y\subseteq\P_{k}^{n}$, el cono afín sobre $Y$ es
    \begin{equation*}
        C(Y)=\{(x_{0},\dots,x_{n})\in\A_{k}^{n+1}:[x_{0}:\dots:x_{n+1}]\in Y\}\cup\{0\}
    \end{equation*}
\end{dfn}

\begin{lema}
    Sea $I\subseteq k[x_{0},\dots,x_{n}]$ un ideal homogéneo distinto de $\m$. Tomar $Y=V^{p}(I)$, 
    luego
    \begin{equation*}
        C(Y)=V(I)\subseteq\A_{k}^{n+1}
    \end{equation*}
\end{lema}

\begin{proof}
    Sea $p\in C(Y)$ no nulo, por definición $[p]\in Y$, entonces $f(p)=0$ para todo 
    $f\in I$ homogéneo. Sea $f\in I$, escribimos
    \begin{equation*}
        f=\sum_{r}f_{r} \htext{con $f_{r}$ homogéneo}
    \end{equation*}
    Como $I$ es ideal homogéneo $f_{r}\in I$, entonces $f_{r}(p)=0$ y por lo tanto $f(p)=0$, luego
    $p\in V(I)$.

    \vspace{1mm}
    Sea $p\in V(I)$ no nulo, por lo tanto $f(p)=0$ para todo $f\in I$, en particular, para todo 
    $f\in I$ homogéneo y por ende $[p]\in Y$, es decir, $p\in C(Y)$.
\end{proof}

\begin{lema}
    Sea $I\subseteq k[x_{0},\dots,x_{n}]$ un ideal homogéneo distinto de $\m$. Consideremos 
    $Y=V^{p}(I)$, entonces $I(C(Y))=I^{h}(Y)$.
\end{lema}

\begin{proof}
    Sea $f\in I(C(Y))$, entonces $f|_{C(Y)}\equiv0$. Digamos que
    \begin{equation*}
        f=\sum_{r}f_{r} \htext{con $f_{r}$ homogéneo}
    \end{equation*}
    Queremos probar que $f_{r}|_{Y}\equiv0$ para todo $r$. Dado $[p]\in Y$ notamos que 
    $p\in C(Y)$, lo que implica que $f(p)=0$, más aún, $f(\lambda p)=0$ para todo $\lambda\neq0$.
    Entonces el polinomio
    \begin{equation*}
        g(\lambda)=f(\lambda p)=\sum_{r}f_{r}(p)\lambda^{r}
    \end{equation*}
    tiene infinitas raíces, lo que implica que $g$ es el polinomio nulo. Luego, $f_{r}(p)=0$ para 
    todo $r$. Concluimos que $f\in I^{h}(Y)$.

    \vspace{1mm}
    Dado $f\in I^{h}(Y)$, digamos que
    \begin{equation*}
        f=\sum_{r}f_{r} \htext{con $f_{r}$ homogéneo y}f_{r}|_{Y}\equiv0
    \end{equation*}
    Fijemos $p\in C(Y)$ no nulo, nuevamente, por definición tenemos que $[p]\in Y$ y se sigue que 
    $f_{r}(p)=0$ para todo $r$, por lo tanto $f(p)=0$. En resumen, $f\in I(C(Y))$.
\end{proof}

\begin{cor}
    Sea $I\subseteq k[x_{0},\dots,x_{n}]$ un ideal homogéneo e $Y=V^{p}(I)$, entonces $Y$ es
    irreducible si y solo si $C(Y)$ es irreducible.
\end{cor}

\begin{teo}[\textbf{Nullstellensatz Proyectivo}]
    Existe una biyección, que invierte inclusiones, entre los conjuntos
    
    \begin{equation*}
        \begin{Bmatrix}
            \text{Variedades proyectivas} \\
            \text{en }\P_{k}^{n} \text{ no vacías}
        \end{Bmatrix}
        \longleftrightarrow
        \begin{Bmatrix}
            \text{Ideales homogéneos radicales} \\
            \text{en }k[x_{0},\dots,x_{n}] \text{ que no} \\
            \text{contienen a }\m
        \end{Bmatrix}
    \end{equation*}
    Recordemos que $\m=(x_{0},\dots,x_{n})$.
\end{teo}

\begin{proof}
    Sea $I\subseteq k[x_{0},\dots,x_{n}]$ un ideal homogéneo radical que no contiene a $\m$.
    Consideremos $Y=V^{p}(I)$, luego
    \begin{equation*}
        I^{h}(Y)=I(C(Y))=I(V(I))=\sqrt{I}=I
    \end{equation*}
    en la penúltima igualdad usamos Nullstellensatz afín. Sea $V\subseteq\P_{k}^{n}$ una variedad 
    proyectiva. Es inmediato que
    \begin{equation*}
        V\subseteq V^{p}(I^{h}(V))
    \end{equation*}
    Por otro lado, sea $J\subseteq k[x_{0},\dots,x_{n}]$ un ideal homogéneo tal que $V=V^{p}(J)$,
    como todo elemento en $J$ se anula en $V$, tenemos que
    \begin{equation*}
        J\subseteq I^{h}(V) \hhtext{entonces} V^{p}(I^{h}(V))\subseteq V^{p}(J)=V
    \end{equation*}
    Por lo tanto $V=V^{p}(I^{h}(V))$.
\end{proof}

\noindent Por último definimos las cuasi variedades proyectivas, que revisitaremos más adelante

\begin{dfn}
    Un conjunto $V\subseteq\P_{k}^{n}$ se dice cuasi variedad proyectiva si existen $U$ abierto y
    $Z$ cerrado tales que $V=U\cap Z$.
\end{dfn}

\begin{ejm}
    Los abiertos y cerrados, esto último equivale a una variedad proyectiva, en $\P_{k}^{n}$ son
    cuasi variedades proyectivas. El conjunto $V^{p}(x^{2}+y^{2}+z^{2})\cap U_{0}$ es una cuasi 
    variedad proyectiva.
\end{ejm}

\noindent Recordemos la inclusión
\begin{align*}
    \phi_{0}:\A_{k}^{n} &\to U_{0} \\
    (a_{1},\dots,a_{n}) &\mapsto [1:a_{1}:\dots:a_{n}]
\end{align*}
Para una variedad afín $V\subseteq\A_{k}^{n}$ consideramos $V_{0}:=\phi_{0}(V)$.

\begin{dfn}
    Sea $V\subseteq\A_{k}^{n}$ una variedad afín. La clausura proyectiva de $V$ se define como
    \begin{equation*}
        \overline{V}:=\overline{V_{0}}\subseteq\P_{k}^{n}
    \end{equation*}
    donde la clausura es respecto a la topología de Zariski.
\end{dfn}

\noindent Usualmente se piensa en $V$ bajo la imagen de la inclusión $\phi_{0}$, es decir, como 
$V_{0}$. Por definición, la clausura proyectiva de $V$ es la variedad proyectiva mas pequeña, en 
términos de contención, que contiene a $V$.

\begin{ejm}
    Consideremos $V=V(y-x^{2})\subseteq\A_{k}^{n}$. Si consideramos el polinomio homogéneo 
    $yz-x^{2}$ vemos que $V\subseteq V^{p}(yz-x^{2})$.

    \vspace{1mm}
    \noindent Por otro lado, sea $W\subseteq\P_{k}^{n}$ variedad proyectiva tal que 
    $V\subseteq W$. Tomemos $f\in k[x_{0},\dots,x_{n}]$ homogéneo tal que $f|_{W}\equiv0$. 
    Consideremos
    \begin{equation*}
        g(x,y)=f(1,x,y) \hhtext{y notar que} f(1,p)=0 \text{ para todo } p\in V\subseteq\A_{k}^{n}
    \end{equation*}
    Equivalentemente, tenemos que $g\in I(V)=(y-x^{2})$ y por ende $y-x^{2}\mid g$. Entonces
    \begin{equation*}
        yz-x^{2}\mid z^{deg(g)}g(x/z,y/z)\mid f
    \end{equation*}
    Así, dado $p\in V^{p}(yz-x^{2})$ se tiene que $f(p)=0$ porque $yz-x^{2}\mid f$. Luego,
    \begin{equation*}
        I^{h}(W)\subseteq I^{h}(V^{p}(yz-x^{2})) \hhtext{lo que implica que}
        V^{p}(yz-x^{2})\subseteq W
    \end{equation*}
    Concluimos que $\overline{V}=V^{p}(yz-x^{2})$.
\end{ejm}

\noindent Intuitivamente, el conjunto $V^{p}(yz-x^{2})\setminus V$ consiste de un único punto ``en 
el infinito`` que corresponde a $[0:0:1]$, por lo que el cerrado más pequeño que contiene a $V$ 
debe ser $V^{p}(yz-x^{2})$.

\begin{dfn}

    \begin{equation*}

\end{dfn}

\subsection{Morfismos de (Cuasi) Variedades Proyectivas}

\noindent Del mismo modo que en el caso afín, nos gustaría crear la categoría de variedades 
proyectivas y cuasi variedades proyectivas. Ya tenemos los objetos, faltan los morfismos, y ese es 
el objetivo de esta parte.

\begin{dfn}
    Sean $V\subseteq\P_{k}^{n}$ y $W\subseteq\P_{k}^{m}$ variedades proyectivas. Un mapa 
    $f:V\to W$ es un mapa homogéneo si existen polinomios homogéneos $f_{0},\dots,f_{m}$ del mismo
    grado tales que
    \begin{equation*}
        f([x])=[f_{0}(x):\dots:f_{m}(x)]
    \end{equation*}
\end{dfn}

\noindent Bajo esta definición los posibles isomorfismos entre variedades proyectivas son 
limitados, más aún, se reducen a los siguientes

\begin{dfn}
    Decimos que $V$ y $W$ son proyectivamente equivalentes si existe un cambio de coordenadas $f$ 
    en $\P_{k}^{n}$, esto es, un mapa polinomial invertible tal que $deg(f_{i})=1$ para todo $i$.
\end{dfn}

\begin{prop}
    Los automorfismos, bajo equivalencia proyectiva, se describen como
    \begin{equation*}
        Aut(\P_{k}^{n})\cong PGL(n+1,k)=GL(n+1,k)/Z(GL(n+1,k))
    \end{equation*}
\end{prop}

\noindent De este modo obtuvimos una categoría $\textbf{ProyVar}_{naive}$, donde los morfismos son 
mapas polinomiales globales y los isomorfismos están dados por equivalencias proyectivas. Sin 
embargo, no es del todo satisfactoria por sus restricciones. Mejoraremos esta definición

\begin{dfn}
    Sean $V\subseteq\P_{k}^{n}$ y $W\subseteq\P_{k}^{m}$ variedades proyectivas. Decimos que una 
    aplicación $\varphi:V\to W$ es morfismo de variedades proyectivas si para todo punto 
    $p\in \P_{k}^{n}$
    \begin{equation*}
        \text{existe una vecindad } U\subseteq V 
        \text{ y polinomios homogéneos del mismo grado }f_{0},\dots,f_{m}\in k[x_{0},\dots,x_{n}]
    \end{equation*}
    tales que $f:U\to W$ esta dado por $f([x])=[f_{0}(x):\dots:f_{m}(x)]$
\end{dfn}

\begin{ejm}
    Consideremos el morfismo
    \begin{equation*}
        \varphi:\P_{k}^{1} \to V(xz-y^{2})\subseteq\P_{k}^{2}
        \hhtext{dado por} [s:t] \mapsto [s^{2}:st:t^{2}]
    \end{equation*}
    y notemos que si $s\neq0$, entonces proyectando respecto a las primeras dos coordenadas 
    obtenemos $[s^{2}:st]=[s:t]$. Así, la inversa viene dada por
    \begin{align*}
        \psi:V(xz-y^{2}) &\to \P_{k}^{1} \\
        [x:y:z] &\mapsto 
        \begin{cases}
            [x:y] \quad\text{si }x\neq0 \\
            [y:z] \quad\text{si }z\neq0
        \end{cases}
    \end{align*}
    El mapa esta bien definido, en efecto, si $x\neq0$ y $z\neq0$ tenemos que
    \begin{equation*}
        [x:y]=[xz:yz]=[y^{2}:yz]=[y:z]
    \end{equation*}
    Luego, sea $[x:y:z]\in V(xz-y^{2})$ y supongamos, sin perdidad de generalidad, que $x\neq0$, 
    entonces
    \begin{equation*}
        \varphi\circ\psi([x:y:z])=\varphi([x:y])=[x^{2}:xy:y^{2}]=[x^{2}:xy:xz]=[x:y:z]
    \end{equation*}
    Por otro lado, dado $[s:t]\in\P_{k}^{1}$ tal que $t\neq0$ vemos que
    \begin{equation*}
        \psi\circ\varphi([s:t])=\psi([s^{2}:st:t^{2}])=[st:t^{2}]=[s:t]
    \end{equation*}
    Concluimos que $\P_{k}^{1}\cong V(xz-y^{2})$ como variedades proyectivas.
\end{ejm}

\newpage
\noindent Sea $\varphi:V\to W$ un morfismo de variedades proyectivas, dado $x\in V$ existen 
$U_{x}\subseteq V$ vecindad de $x$ y polinomios $f_{0},\dots,f_{m}$ tales que
\begin{equation*}
    \varphi:U\to W \htext{esta dado por } \varphi(x)=[f_{0}(x):\dots:f_{m}(x)]
\end{equation*}
Sea $V'\subseteq W$ una variedad proyectiva, entonces $V'=V^{p}(T)$ para algún conjunto $T$ de 
polinomios homogéneos. Luego, vemos que
\begin{equation*}
    \varphi^{-1}(V')\cap U_{x}=V^{p}(\{g(f_{0},\dots,f_{m}):g\in T\})\cap V\cap U_{x}
\end{equation*}
y por lo tanto $\varphi^{-1}(V')\cap U_{x}$ es cerrado en $U_{x}$. Consideremos la colección 
$\{U_{x}\}_{x\in V}$, que es un cubrimiento abierto de $V$.

\begin{lema}
    Sea $\varphi:V\to W$ un morfismo de variedades proyectivas, entonces $\varphi$ es continuo
    en la topología Zariski.
\end{lema}

\begin{proof}
    Sea $V'\subseteq W$ una variedad proyectiva. Luego, $\varphi^{-1}(V')$ es cerrado en $V$ si y
    solo si $\O=V\setminus\varphi^{-1}(V')$ es abierto en $V$. Notamos que
    \begin{equation*}
        \O=\O\cap V=\O\cap\left(\bigcup_{x\in V}U_{x}\right)=\bigcup_{x\in V}(\O\cap U_{x})
    \end{equation*}
    Basta probar que $\O\cap U_{x}$ es abierto en $U_{x}$, ya que $U_{x}$ es abierto en $V$. Así,
    \begin{align*}
        U_{x}\setminus(\O\cap U_{x})=U_{x}\setminus\O=U_{x}\setminus(V\setminus \varphi^{-1}(V'))
        =\varphi^{-1}(V')\cap U_{x}
    \end{align*}
    lo que concluye la demostración, puesto que $\varphi^{-1}(V')\cap U_{x}$ es cerrado en 
    $U_{x}$.
\end{proof}

\begin{prop}
    Sean $\varphi:V\to W$ y $\psi:W\to Z$ morfismos de variedades proyectivas, entonces 
    $\psi\circ\varphi:V\to Z$ es morfismo de variedades proyectivas.
\end{prop}

\begin{proof}
    Sea $x\in V$, existen $U_{x}\subseteq V$ y $U_{f(x)}\subseteq W$ abiertos tales que
    \begin{equation*}
        \varphi:U_{x}\to W \htext{esta dado por } \varphi([x])=[f_{0}(x):\dots:f_{m}(x)]
    \end{equation*}
    y
    \begin{equation*}
        \psi:U_{f(x)}\to W \htext{esta dado por } \psi([x])=[g_{0}(x):\dots:g_{l}(x)]
    \end{equation*}
    para algunos polinomios homogéneos. Consideremos $U=U_{x}\cap\varphi^{-1}(U_{f(x)})$ que es
    abierto no vacío en $V$ porque $\varphi$ es continua. De este modo,
    \begin{align*}
        \varphi\circ\psi:U &\to Z \htext{esta dado por } \\
        x &\mapsto [g_{0}(f_{0}(x),\dots,f_{m}(x)):\dots:g_{l}(f_{0}(x),\dots,f_{m}(x))]
    \end{align*}
    Lo que prueba el resultado.
\end{proof}

\section{Teoría de la Dimensión} %% Clase 6 y 7

\begin{lema}
    Sea $f:X\to Y$ un morfismo de variedades sobreyectivo, entonces $dim(Y)\leq dim(X)$.
\end{lema}

\begin{proof}
    Sea $Y_{j}$ componente irreducible de dimensión maximal, es decir, $dim(Y_{j})=dim(Y)$. 
    Tomando la restricción $f:f^{-1}(Y_{j})\to Y_{j}$ podemos asumir que $Y_{j}$ es irreducible.
    Veamos que
    \begin{equation*}
        Y=f\left(\bigcup X_{i}\right)=\overline{\bigcup f(X_{i})}=\bigcup\overline{f(X_{i})}
    \end{equation*}
    donde la última igualdad se debe a que la unión es finita. Como $Y$ es irreducible, tenemos 
    que $Y=\overline{f(X_{j})}$ para algún $j$. 
    
    Así, podemos asumir que $f:X\to Y$ es morfismo entre variedades irreducibles tal que 
    $\overline{f(X)}=Y$. Veamos que
    \begin{equation*}
        f^{*}:k[Y]\to k[X]
    \end{equation*}
    es inyectivo. Sea $g\in\kr{f^{*}}$, entonces $(g\circ f)(x)=0$ para todo $x\in X$, dicho de 
    otro modo,
    \begin{equation*}
        f(X)\subseteq V(g) \hhtext{entonces} Y=\overline{f(X)}\subseteq V(g)
    \end{equation*}
    concluimos que $g|_{Y}\equiv0$. Por lo tanto $k(Y)\subseteq k(X)$ y entonces
    \begin{equation*}
        dim(Y)=trdeg_{k}(k(Y))\leq trdeg_{k}(k(X))=dim(X)
    \end{equation*}
\end{proof}
